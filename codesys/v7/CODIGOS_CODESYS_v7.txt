==================== E_ESTADO ====================
// DUT (Enumeración) — Estados de la máquina de estados del secador
// Crear en CODESYS: Application > Agregar objeto > DUT > Enumeración, nombre E_ESTADO
{attribute 'qualified_only'}
{attribute 'strict'}
TYPE E_ESTADO :
(
    REPOSO       := 0,   // Listo, esperando START (amarilla fija)
    CARGA        := 1,   // Bandeja entra, compuerta a posición normal
    SECADO       := 2,   // Ventilador + resistencias, perfil térmico
    INVERSION    := 3,   // Cambio de sentido del aire (cada 6 h de lote)
    ENFRIAMIENTO := 4,   // Ventilador sin resistencia
    FIN_SECADO   := 5,   // Lote seco, esperando START para descargar (verde intermitente)
    DESCARGA     := 6,   // Bandeja sale
    PARO         := 7,   // Pausa por STOP (amarilla fija)
    FALLA        := 8,   // Comunicación o cilindro trabado (roja intermitente)
    EMERGENCIA   := 9    // Pulsador de emergencia (roja fija)
);
END_TYPE

==================== GVL_SECUENCIA ====================
// GVL_SECUENCIA v7 (19 oct 2026, sin cambios respecto a v6) — Variables compartidas entre FluidSIM (OPC), la HMI,
// el programa SECUENCIA y PLC_PRG.
// Todas deben estar marcadas en "Symbol configuration" para que el
// servidor OPC de CODESYS las publique a FluidSIM.
//
// Cambios v6:
//   - sin_fluidsim: TRUE = el PLC simula pulsadores de campo y cilindros (pruebas sin FluidSIM).
//   - lote_preparado arranca en TRUE: el server recién iniciado ya es un lote nuevo,
//     así que START funciona sin pulsar antes "Nuevo lote".
//
// Intercambio con FluidSIM por OPC: un BYTE por módulo de 8 bits.
//   INPUT0  <- bloque "FluidSIM Out" (pulsadores, finales de carrera) -> se lee INPUT0.n
//   OUTPUT0 -> bloque "FluidSIM In"  (bobinas, relé, lámparas)        -> se escribe OUTPUT0.n
// En SECUENCIA se copian bit a bit a las variables I_ / Q_ con nombre.
VAR_GLOBAL
    // ===== Modo de prueba (v6) =====
    sin_fluidsim    : BOOL := TRUE;   // TRUE = sin FluidSIM (el PLC simula); FALSE = todo viene de FluidSIM

    // ===== Módulo 0 de FluidSIM (OPC) =====
    INPUT0          : BYTE;   // desde FluidSIM Out (módulo 1, puerto 1)
    OUTPUT0         : BYTE;   // hacia FluidSIM In  (módulo 1, puerto 1)

    // ===== Entradas con nombre (copia de INPUT0.n, o simuladas si sin_fluidsim) =====
    I_EMERGENCIA_NC : BOOL;   // INPUT0.0  Paro de emergencia, contacto NC (TRUE = normal, FALSE = pulsado)
    I_START         : BOOL;   // INPUT0.1  Pulsador START, NA
    I_STOP_NC       : BOOL;   // INPUT0.2  Pulsador STOP, contacto NC (TRUE = normal, FALSE = pulsado)
    I_1B1           : BOOL;   // INPUT0.3  Cilindro 1 (compuerta) atrás   -> aire normal (abajo -> arriba)
    I_1B2           : BOOL;   // INPUT0.4  Cilindro 1 (compuerta) adelante -> aire invertido
    I_2B1           : BOOL;   // INPUT0.5  Cilindro 2 (bandeja) atrás   -> bandeja afuera
    I_2B2           : BOOL;   // INPUT0.6  Cilindro 2 (bandeja) adelante -> bandeja adentro

    // ===== Salidas con nombre (se copian a OUTPUT0.n) =====
    Q_1S1           : BOOL;   // OUTPUT0.0 Bobina: cilindro 1 sale (compuerta -> invertida)
    Q_2S1           : BOOL;   // OUTPUT0.1 Bobina: cilindro 1 entra (compuerta -> normal)
    Q_1S2           : BOOL;   // OUTPUT0.2 Bobina: cilindro 2 sale (bandeja entra al secador)
    Q_2S2           : BOOL;   // OUTPUT0.3 Bobina: cilindro 2 entra (bandeja sale del secador)
    Q_K1            : BOOL;   // OUTPUT0.4 Relé K1: ventilador ON/OFF
    Q_H1_VERDE      : BOOL;   // OUTPUT0.5 Ciclo en marcha (fija) / fin de secado (intermitente)
    Q_H2_AMARILLA   : BOOL;   // OUTPUT0.6 Reposo o paro
    Q_H3_ROJA       : BOOL;   // OUTPUT0.7 Emergencia (fija) / falla (intermitente)

    // ===== HMI (botones con "Pulsar", NO "Alternar") =====
    hmi_start       : BOOL;
    hmi_stop        : BOOL;
    hmi_nuevo_lote  : BOOL;   // botón "Nuevo lote"

    // ===== Estado (para la HMI) =====
    estado          : E_ESTADO := E_ESTADO.REPOSO;
    estado_txt      : STRING(30) := 'REPOSO';
    falla_codigo    : INT;    // 0 = sin falla, 1 = comunicación Python, 2 = compuerta trabada,
                              // 3 = bandeja trabada, 4 = finales de carrera incoherentes
    falla_txt       : STRING(40);

    // ===== Interfaz SECUENCIA -> PLC_PRG (las escribe SOLO SECUENCIA) =====
    ventilador      : BOOL;   // pide ventilador encendido
    calentar        : BOOL;   // permiso de resistencias (PLC_PRG aplica el lazo térmico)
    planta_run      : BOOL;   // TRUE -> plant_mode = 0 (RUN); FALSE -> 1 (HOLD)
    aire_invertido  : BOOL;   // sentido del aire = posición REAL de la compuerta (1B2)
    nuevo_lote      : BOOL;   // pulso de un ciclo al pulsar "Nuevo lote" (PLC_PRG reinicia Python)
    lote_preparado  : BOOL := TRUE;   // v6: TRUE al arrancar; START solo arranca si está en TRUE
END_VAR

==================== PLC_PRG ====================
// PLC_PRG v7 (19 oct 2026): v6 + selección de PERFIL TÉRMICO y CAUDAL en la HMI.
//   - perfil_sel (HMI): 0 = REF 50 °C, 1 = P1 45 °C, 2 = P2 55 -> 35 °C, 3 = P3 35 -> 55 °C.
//     El cambio de P2/P3 ocurre cuando la humedad PROMEDIO del lote llega a 30 % b.h.
//     (de Oliveira et al., 2018; diseño en python/perfiles_termicos.py).
//   - caudal_sel (HMI): 30, 75 o 100 %.
//   - Ambos se toman SOLO en REPOSO; al pulsar START quedan fijos todo el lote (perfil_activo,
//     caudal_activo), como en el diseño experimental.
//   - El PLC calcula la consigna y la envía a Python (registro 26, setpoint_cmd), que la usa
//     en el termostato. Así la consigna de la HMI es la que se aplica de verdad.
// SECUENCIA se llama DESPUÉS de PLC_PRG en la MainTask.
PROGRAM PLC_PRG
VAR
    // --- Crudo Modbus (words) que se mapean en IO Mapping ---
    T_process_raw   : ARRAY[0..1] OF WORD;
    RH_ambient_raw  : ARRAY[0..1] OF WORD;
    heater_cmd      : WORD;
    fan_cmd         : WORD;
    plant_mode      : WORD;
    safety_ok       : WORD;

    // --- Conversión a REAL ---
    conv_T          : U_WORDS_TO_REAL;
    conv_RH         : U_WORDS_TO_REAL;
    T_process       : REAL;
    RH_ambient      : REAL;

    // --- Auxiliares para el HMI ---
    heater_on       : BOOL;
    fan_on          : BOOL;
    safety_alarm    : BOOL;
    planta_on       : BOOL;   // piloto "Planta en funcionamiento"
    aire_invertido  : BOOL;   // piloto HMI: sentido del aire invertido

    setpoint        : REAL := 50.0;   // °C, consigna aplicada (v7: la calcula el perfil; REF = 50 °C, González et al., 2010)
    tiempo_proceso_raw : ARRAY[0..1] OF WORD;  // [0]=tiempo_proceso_hi, [1]=tiempo_proceso_lo
    conv_tiempo         : U_WORDS_TO_REAL;
    tiempo_secado_h     : REAL;

    M_coffee_raw  : ARRAY[0..1] OF WORD;
    conv_mcoffee  : U_WORDS_TO_REAL;
    M_coffee      : REAL;

    heartbeat_actual   : WORD;   // viene del IO Mapping
    heartbeat_anterior : WORD;
    tmr_comm           : TON;
    comm_error         : BOOL;

    RH_process_raw  : ARRAY[0..1] OF WORD;   // [0] <- RH_process_hi (dir 13), [1] <- RH_process_lo (dir 14)
    conv_RH_process : U_WORDS_TO_REAL;
    RH_process      : REAL;                   // % HR dentro de la cámara

    // ===== NUEVO (4 oct 2026) =====
    flow_dir_cmd      : WORD;                  // canal FC06, dir 15
    M_coffee_max_raw  : ARRAY[0..1] OF WORD;   // [0] <- dir 16 (_hi), [1] <- dir 17 (_lo)
    conv_mcoffee_max  : U_WORDS_TO_REAL;
    M_coffee_max      : REAL;                  // % b.h. de la capa más húmeda
    heater_duty_raw   : ARRAY[0..1] OF WORD;   // [0] <- dir 18 (_hi), [1] <- dir 19 (_lo)
    conv_duty         : U_WORDS_TO_REAL;
    heater_duty       : REAL;                  // % del ciclo con la resistencia encendida

    // ===== NUEVO (5 oct 2026): manipulada 2, caudal variable =====
    fan_speed_pct     : WORD := 100;           // canal FC06, dir 20: consigna de caudal 30-100 % (HMI)
    caudal_raw        : ARRAY[0..1] OF WORD;   // [0] <- dir 22 (_hi), [1] <- dir 23 (_lo)
    conv_caudal       : U_WORDS_TO_REAL;
    caudal_m3_min     : REAL;                  // caudal real aplicado (8.0 = 100 %)

    // ===== NUEVO v5 (14 oct 2026) =====
    caudal_sel        : INT := 100;            // selector de la HMI (cuadro combinado): 30, 75 o 100 %
    reset_lote_cmd    : WORD;                  // canal FC06, dir 21: pulso "Nuevo lote" hacia Python
    tp_reset          : TP;                    // mantiene el pulso 3 s para que Python lo vea
    energia_raw       : ARRAY[0..1] OF WORD;   // [0] <- dir 24 (_hi), [1] <- dir 25 (_lo)
    conv_energia      : U_WORDS_TO_REAL;
    energia_kWh       : REAL;                  // energía eléctrica del lote (resistencia + ventilador)

    // ===== NUEVO v7 (19 oct 2026): perfiles térmicos =====
    perfil_sel        : INT := 0;              // selector HMI: 0 REF, 1 P1, 2 P2, 3 P3
    perfil_activo     : INT := 0;              // copia fija durante el lote
    caudal_activo     : INT := 100;            // copia fija durante el lote
    sp_inicial        : REAL := 50.0;          // °C antes del cambio
    sp_final          : REAL := 50.0;          // °C después del cambio (30 % b.h.)
    cambio_perfil     : BOOL;                  // TRUE cuando el promedio ya bajó de 30 % b.h. (enclavado)
    setpoint_cmd      : WORD;                  // canal FC06, dir 26: consigna hacia Python (°C enteros)
    perfil_txt        : STRING(40);            // texto para la HMI

    lazo_local        : BOOL := TRUE;          // TRUE = termostato en Python (demo acelerada)
                                               // FALSE = el PLC conmuta la resistencia (solo con --factor-aceleracion 1 --lazo-termico plc)
    fin_secado        : BOOL;                  // se enclava al llegar a la humedad objetivo
    reset_lote        : BOOL;                  // botón HMI: nuevo lote (borra fin_secado)
                                               // v4: también lo borra GVL_SECUENCIA.nuevo_lote
    en_marcha         : BOOL;
END_VAR

// ================== CONVERSIONES (palabra BAJA en words[0]) ==================
conv_T.words[0] := T_process_raw[1];
conv_T.words[1] := T_process_raw[0];
T_process := conv_T.value;

conv_RH.words[0] := RH_ambient_raw[1];
conv_RH.words[1] := RH_ambient_raw[0];
RH_ambient := conv_RH.value;

conv_tiempo.words[0] := tiempo_proceso_raw[1];
conv_tiempo.words[1] := tiempo_proceso_raw[0];
tiempo_secado_h := conv_tiempo.value;

conv_mcoffee.words[0] := M_coffee_raw[1];
conv_mcoffee.words[1] := M_coffee_raw[0];
M_coffee := conv_mcoffee.value;

conv_RH_process.words[0] := RH_process_raw[1];
conv_RH_process.words[1] := RH_process_raw[0];
RH_process := conv_RH_process.value;

conv_mcoffee_max.words[0] := M_coffee_max_raw[1];
conv_mcoffee_max.words[1] := M_coffee_max_raw[0];
M_coffee_max := conv_mcoffee_max.value;

conv_duty.words[0] := heater_duty_raw[1];
conv_duty.words[1] := heater_duty_raw[0];
heater_duty := conv_duty.value;

conv_caudal.words[0] := caudal_raw[1];
conv_caudal.words[1] := caudal_raw[0];
caudal_m3_min := conv_caudal.value;

conv_energia.words[0] := energia_raw[1];   // v5
conv_energia.words[1] := energia_raw[0];
energia_kWh := conv_energia.value;

// ================== PERFIL Y CAUDAL (v7) ==================
// Se leen de la HMI solo en REPOSO; durante el lote quedan fijos.
IF GVL_SECUENCIA.estado = E_ESTADO.REPOSO THEN
    IF (perfil_sel >= 0) AND (perfil_sel <= 3) THEN
        perfil_activo := perfil_sel;
    END_IF
    caudal_activo := caudal_sel;
    cambio_perfil := FALSE;
END_IF

CASE perfil_activo OF
    0: sp_inicial := 50.0; sp_final := 50.0; perfil_txt := 'REF: 50 C constante';
    1: sp_inicial := 45.0; sp_final := 45.0; perfil_txt := 'P1: 45 C constante';
    2: sp_inicial := 55.0; sp_final := 35.0; perfil_txt := 'P2: 55 C -> 35 C al 30 % b.h.';
    3: sp_inicial := 35.0; sp_final := 55.0; perfil_txt := 'P3: 35 C -> 55 C al 30 % b.h.';
END_CASE

// Cambio al 30 % b.h. (humedad promedio). Se enclava: no vuelve atrás en el mismo lote.
// M_coffee > 0 evita disparar con el registro aún en 0.0 al arrancar.
IF (M_coffee > 0.0) AND (M_coffee <= 30.0) AND (GVL_SECUENCIA.estado <> E_ESTADO.REPOSO) THEN
    cambio_perfil := TRUE;
END_IF

IF cambio_perfil THEN
    setpoint := sp_final;
ELSE
    setpoint := sp_inicial;
END_IF
setpoint_cmd := REAL_TO_WORD(setpoint);

// v5/v7: consigna de caudal fija del lote
fan_speed_pct := INT_TO_WORD(caudal_activo);
// Consigna de caudal limitada a 30-100 % (mínimo de operación del ventilador)
IF fan_speed_pct > 100 THEN fan_speed_pct := 100; END_IF
IF fan_speed_pct < 30 THEN fan_speed_pct := 30; END_IF

// ================== WATCHDOG (heartbeat de Python) ==================
IF heartbeat_actual <> heartbeat_anterior THEN
    tmr_comm(IN := FALSE);
ELSE
    tmr_comm(IN := TRUE, PT := T#4S);
END_IF
heartbeat_anterior := heartbeat_actual;   // DESPUÉS de comparar
comm_error := tmr_comm.Q;

// ================== NUEVO LOTE (v5) ==================
// SECUENCIA pone nuevo_lote un ciclo al pulsar el botón; el TP lo alarga a 3 s
// para que Python (periodo 1 s) vea el flanco 0 -> 1 en el registro 21.
tp_reset(IN := GVL_SECUENCIA.nuevo_lote, PT := T#3S);
reset_lote_cmd := BOOL_TO_WORD(tp_reset.Q);

// ================== FIN DE SECADO (enclavado) ==================
// Promedio <= 11 % b.h. Y capa más húmeda <= 12 % b.h. (SECAFÉ, 2008).
// M_coffee > 0 evita disparar con el registro aún en 0.0 al arrancar.
// v5: no se enclava mientras dura el pulso de nuevo lote (Python aún publica el lote anterior).
IF (NOT tp_reset.Q) AND (M_coffee > 0.0) AND (M_coffee <= 11.0)
   AND (M_coffee_max > 0.0) AND (M_coffee_max <= 12.0) THEN
    fin_secado := TRUE;
END_IF
IF reset_lote OR tp_reset.Q THEN   // v5
    fin_secado := FALSE;
    reset_lote := FALSE;
END_IF

// ================== SUPERVISIÓN (v4) ==================
// La máquina de estados (SECUENCIA) decide CUÁNDO hay ventilador, resistencia y RUN.
// PLC_PRG solo agrega la seguridad de comunicación y el lazo térmico.
plant_mode := BOOL_TO_WORD(NOT GVL_SECUENCIA.planta_run);   // 0 = RUN, 1 = HOLD
en_marcha  := GVL_SECUENCIA.ventilador AND NOT comm_error;

fan_cmd := BOOL_TO_WORD(en_marcha);

IF lazo_local THEN
    // Python regula a 50 °C cada segundo de modelo; el PLC solo da el permiso.
    heater_cmd := BOOL_TO_WORD(en_marcha AND GVL_SECUENCIA.calentar);
ELSE
    // Todo/nada en el PLC: válido SOLO a tiempo real (factor 1).
    heater_cmd := BOOL_TO_WORD(en_marcha AND GVL_SECUENCIA.calentar AND (T_process < setpoint));
END_IF

// v4: el sentido del aire sale de la POSICIÓN REAL de la compuerta (final de carrera 1B2
// en FluidSIM), no del reloj. El reloj de 6 h (SECAFÉ, 2008) está ahora en SECUENCIA.
flow_dir_cmd := BOOL_TO_WORD(GVL_SECUENCIA.aire_invertido AND NOT comm_error);

// ================== AUXILIARES HMI ==================
IF lazo_local THEN
    heater_on := en_marcha AND (heater_duty > 0.5);
ELSE
    heater_on := (heater_cmd = 1);
END_IF
fan_on       := (fan_cmd = 1);
safety_alarm := (safety_ok = 0) OR comm_error;
planta_on    := en_marcha;
aire_invertido := (flow_dir_cmd = 1);

==================== SECUENCIA ====================
// =====================================================================
// SECUENCIA — Máquina de estados del secador de café (CASE OF)
// Proyecto: Supervisión IoT del secado de café (CODESYS V3.5 SP9)
// Versión: v6, 19 oct 2026
//   v6: modo sin_fluidsim (el PLC simula pulsadores de campo y cilindros).
//   14 oct: botón Nuevo lote + auxiliares HMI.
//
// Llamar en la MainTask DESPUÉS de PLC_PRG:
//   MainTask: 1) PLC_PRG   (lee Python por Modbus, watchdog, fin_secado)
//             2) SECUENCIA (decide estado, cilindros, ventilador, semáforo)
//
// Dueño de variables: SECUENCIA escribe solo en GVL_SECUENCIA.
// De PLC_PRG solo LEE: comm_error, safety_ok, fin_secado, tiempo_secado_h.
//
// Fuentes:
// - Inversión del aire cada 6 h: Parra-Coronado et al. (2008), SECAFÉ, Cenicafé.
// - Colores del semáforo (verde normal, amarillo atención, rojo emergencia):
//   IEC 60204-1, apartado de colores de indicadores luminosos.
// =====================================================================
PROGRAM SECUENCIA
VAR
    // --- Mandos ---
    rt_start         : R_TRIG;
    rt_nuevo         : R_TRIG;   // flanco del botón "Nuevo lote"
    start_pulso      : BOOL;   // flanco de START (FluidSIM O HMI)
    stop_activo      : BOOL;   // STOP pulsado (FluidSIM O HMI)
    emergencia       : BOOL;   // emergencia pulsada (solo FluidSIM)

    // --- Diagnóstico ---
    falla_comm       : BOOL;
    falla_fc         : BOOL;   // los dos finales de carrera de un cilindro activos a la vez
    en_ciclo         : BOOL;   // estados donde una falla debe detener el proceso

    // --- Tiempos de lote (horas de MODELO, vienen de Python) ---
    t0_lote_h        : REAL;   // tiempo de modelo al iniciar el secado del lote
    t_lote_h         : REAL;   // horas de secado del lote actual
    t_fin_h          : REAL;   // tiempo de modelo al entrar en ENFRIAMIENTO

    // --- Compuerta de inversión ---
    sentido_pedido      : BOOL;  // TRUE = el reloj de 6 h pide aire invertido
    compuerta_invertida : BOOL;  // posición real: 1B2 activo y 1B1 inactivo
    compuerta_normal    : BOOL;  // posición real: 1B1 activo y 1B2 inactivo

    // --- Órdenes de movimiento (se calculan a partir del estado) ---
    mov_compuerta_inv   : BOOL;
    mov_compuerta_norm  : BOOL;
    mov_bandeja_adentro : BOOL;
    mov_bandeja_afuera  : BOOL;

    // --- Temporizadores (se llaman SIEMPRE, fuera del CASE) ---
    tmr_mov          : TON;    // vigila que el cilindro llegue al final de carrera
    tmr_blink        : TON;    // base de la intermitencia
    parpadeo         : BOOL;

    estado_previo    : E_ESTADO := E_ESTADO.REPOSO;  // para volver después de PARO

    // --- Auxiliares de la HMI (animaciones y textos) ---
    tmr_anim         : TON;
    angulo_vent      : REAL;          // rotor del ventilador
    angulo_compuerta : REAL;          // hoja de la compuerta 0 / 90
    x_bandeja        : INT;           // desplazamiento de la bandeja
    txt_comm         : STRING(10);
    txt_aire         : STRING(30);
    txt_bandeja      : STRING(15);
    txt_vent         : STRING(12);
    txt_resist       : STRING(12);
    txt_alarma       : STRING(50);
    lim_objetivo     : REAL := 11.0;  // línea de referencia en la tendencia
    lim_capa         : REAL := 12.0;  // línea de referencia en la tendencia

    // --- v6: simulación de cilindros cuando sin_fluidsim = TRUE ---
    sim_comp_inv     : BOOL;          // TRUE = compuerta invertida (1B2)
    sim_band_adentro : BOOL;          // TRUE = bandeja adentro (2B2)
END_VAR
VAR CONSTANT
    T_MOV_MAX                 : TIME := T#5S;   // tiempo máximo de carrera de un cilindro (real)
    T_ENFRIAMIENTO_H          : REAL := 0.5;    // h de modelo con ventilador sin resistencia (supuesto propio)
    PERIODO_INVERSION_H       : DINT := 6;      // SECAFÉ (2008)
    PAUSA_RESIST_EN_INVERSION : BOOL := FALSE;  // FALSE en la demo acelerada (factor 540);
                                                // TRUE solo en la verificación a tiempo real
END_VAR

// =====================================================================
// 0. LECTURA DE FLUIDSIM (INPUT0 <- bloque "FluidSIM Out") o SIMULACIÓN (v6)
// =====================================================================
IF GVL_SECUENCIA.sin_fluidsim THEN
    // Pulsadores de campo en reposo: solo manda la HMI
    GVL_SECUENCIA.I_EMERGENCIA_NC := TRUE;
    GVL_SECUENCIA.I_START         := FALSE;
    GVL_SECUENCIA.I_STOP_NC       := TRUE;
    // Cilindros simulados: llegan al final de carrera en cuanto se activa la bobina
    // (usa las bobinas del ciclo anterior)
    IF GVL_SECUENCIA.Q_1S1 THEN sim_comp_inv := TRUE;  END_IF
    IF GVL_SECUENCIA.Q_2S1 THEN sim_comp_inv := FALSE; END_IF
    IF GVL_SECUENCIA.Q_1S2 THEN sim_band_adentro := TRUE;  END_IF
    IF GVL_SECUENCIA.Q_2S2 THEN sim_band_adentro := FALSE; END_IF
    GVL_SECUENCIA.I_1B1 := NOT sim_comp_inv;
    GVL_SECUENCIA.I_1B2 := sim_comp_inv;
    GVL_SECUENCIA.I_2B1 := NOT sim_band_adentro;
    GVL_SECUENCIA.I_2B2 := sim_band_adentro;
ELSE
    GVL_SECUENCIA.I_EMERGENCIA_NC := GVL_SECUENCIA.INPUT0.0;   // EMERGENCY (NC)
    GVL_SECUENCIA.I_START         := GVL_SECUENCIA.INPUT0.1;   // START (NA)
    GVL_SECUENCIA.I_STOP_NC       := GVL_SECUENCIA.INPUT0.2;   // STOP (NC)
    GVL_SECUENCIA.I_1B1           := GVL_SECUENCIA.INPUT0.3;   // compuerta atrás
    GVL_SECUENCIA.I_1B2           := GVL_SECUENCIA.INPUT0.4;   // compuerta adelante
    GVL_SECUENCIA.I_2B1           := GVL_SECUENCIA.INPUT0.5;   // bandeja afuera
    GVL_SECUENCIA.I_2B2           := GVL_SECUENCIA.INPUT0.6;   // bandeja adentro
END_IF
// INPUT0.7 libre

// =====================================================================
// 1. ENTRADAS Y MANDOS
// =====================================================================
GVL_SECUENCIA.nuevo_lote := FALSE;   // pulso de un solo ciclo

rt_start(CLK := GVL_SECUENCIA.I_START OR GVL_SECUENCIA.hmi_start);
start_pulso := rt_start.Q;
rt_nuevo(CLK := GVL_SECUENCIA.hmi_nuevo_lote);

// STOP y EMERGENCIA son NC: el contacto abierto (FALSE) significa pulsado.
// Si el cable se corta o FluidSIM no está conectado, el PLC lo ve como pulsado (falla segura).
stop_activo := (NOT GVL_SECUENCIA.I_STOP_NC) OR GVL_SECUENCIA.hmi_stop;
emergencia  := NOT GVL_SECUENCIA.I_EMERGENCIA_NC;

// Posición real de la compuerta
compuerta_invertida := GVL_SECUENCIA.I_1B2 AND NOT GVL_SECUENCIA.I_1B1;
compuerta_normal    := GVL_SECUENCIA.I_1B1 AND NOT GVL_SECUENCIA.I_1B2;

// Reloj de inversión: cada 6 h de lote cambia el sentido pedido
t_lote_h := PLC_PRG.tiempo_secado_h - t0_lote_h;
IF t_lote_h < 0.0 THEN
    t_lote_h := 0.0;
END_IF
sentido_pedido := ((TRUNC(t_lote_h) / PERIODO_INVERSION_H) MOD 2) = 1;

// =====================================================================
// 2. DIAGNÓSTICO (usa las órdenes de movimiento del ciclo anterior)
// =====================================================================
falla_comm := PLC_PRG.comm_error OR (PLC_PRG.safety_ok = 0);
falla_fc   := (GVL_SECUENCIA.I_1B1 AND GVL_SECUENCIA.I_1B2)
           OR (GVL_SECUENCIA.I_2B1 AND GVL_SECUENCIA.I_2B2);

en_ciclo := (GVL_SECUENCIA.estado = E_ESTADO.CARGA)
         OR (GVL_SECUENCIA.estado = E_ESTADO.SECADO)
         OR (GVL_SECUENCIA.estado = E_ESTADO.INVERSION)
         OR (GVL_SECUENCIA.estado = E_ESTADO.ENFRIAMIENTO)
         OR (GVL_SECUENCIA.estado = E_ESTADO.FIN_SECADO)
         OR (GVL_SECUENCIA.estado = E_ESTADO.DESCARGA)
         OR (GVL_SECUENCIA.estado = E_ESTADO.PARO);

// =====================================================================
// 3. PRIORIDADES: EMERGENCIA > FALLA > PARO (antes del CASE)
// =====================================================================
IF emergencia THEN
    GVL_SECUENCIA.estado := E_ESTADO.EMERGENCIA;

ELSIF en_ciclo AND (falla_comm OR falla_fc OR tmr_mov.Q) THEN
    IF falla_comm THEN
        GVL_SECUENCIA.falla_codigo := 1;
    ELSIF tmr_mov.Q AND (mov_compuerta_inv OR mov_compuerta_norm) THEN
        GVL_SECUENCIA.falla_codigo := 2;
    ELSIF tmr_mov.Q THEN
        GVL_SECUENCIA.falla_codigo := 3;
    ELSE
        GVL_SECUENCIA.falla_codigo := 4;
    END_IF
    GVL_SECUENCIA.estado := E_ESTADO.FALLA;

ELSIF stop_activo AND (
         (GVL_SECUENCIA.estado = E_ESTADO.CARGA)
      OR (GVL_SECUENCIA.estado = E_ESTADO.SECADO)
      OR (GVL_SECUENCIA.estado = E_ESTADO.INVERSION)
      OR (GVL_SECUENCIA.estado = E_ESTADO.ENFRIAMIENTO)
      OR (GVL_SECUENCIA.estado = E_ESTADO.DESCARGA)) THEN
    estado_previo := GVL_SECUENCIA.estado;
    GVL_SECUENCIA.estado := E_ESTADO.PARO;
END_IF

// =====================================================================
// 4. MÁQUINA DE ESTADOS: aquí solo se deciden las TRANSICIONES
// =====================================================================
CASE GVL_SECUENCIA.estado OF

E_ESTADO.REPOSO:
    // Amarilla fija.
    // 1) "Nuevo lote": reinicia la planta en Python (humedad 55 %, tiempo y energía a 0).
    IF rt_nuevo.Q THEN
        GVL_SECUENCIA.nuevo_lote := TRUE;      // PLC_PRG manda el pulso al registro 21
        GVL_SECUENCIA.lote_preparado := TRUE;
        GVL_SECUENCIA.falla_codigo := 0;
    // 2) START: arranca el ciclo solo si hay un lote preparado y todo está sano.
    ELSIF start_pulso AND GVL_SECUENCIA.lote_preparado
          AND NOT stop_activo AND NOT falla_comm AND NOT falla_fc THEN
        GVL_SECUENCIA.estado := E_ESTADO.CARGA;
    END_IF

E_ESTADO.CARGA:
    // Bandeja adentro (2B2) y compuerta en posición normal (1B1)
    IF GVL_SECUENCIA.I_2B2 AND NOT GVL_SECUENCIA.I_2B1 AND compuerta_normal THEN
        t0_lote_h := PLC_PRG.tiempo_secado_h;  // el reloj de inversión empieza en 0
        GVL_SECUENCIA.estado := E_ESTADO.SECADO;
    END_IF

E_ESTADO.SECADO:
    IF PLC_PRG.fin_secado THEN
        t_fin_h := PLC_PRG.tiempo_secado_h;
        GVL_SECUENCIA.estado := E_ESTADO.ENFRIAMIENTO;
    ELSIF sentido_pedido <> compuerta_invertida THEN
        GVL_SECUENCIA.estado := E_ESTADO.INVERSION;
    END_IF

E_ESTADO.INVERSION:
    // Vuelve a SECADO cuando la compuerta llega a la posición pedida
    IF (sentido_pedido AND compuerta_invertida)
       OR (NOT sentido_pedido AND compuerta_normal) THEN
        GVL_SECUENCIA.estado := E_ESTADO.SECADO;
    END_IF

E_ESTADO.ENFRIAMIENTO:
    IF (PLC_PRG.tiempo_secado_h - t_fin_h) >= T_ENFRIAMIENTO_H THEN
        GVL_SECUENCIA.estado := E_ESTADO.FIN_SECADO;
    END_IF

E_ESTADO.FIN_SECADO:
    // Verde intermitente. START = autorizar la descarga.
    IF start_pulso AND NOT stop_activo THEN
        GVL_SECUENCIA.estado := E_ESTADO.DESCARGA;
    END_IF

E_ESTADO.DESCARGA:
    IF GVL_SECUENCIA.I_2B1 AND NOT GVL_SECUENCIA.I_2B2 THEN
        GVL_SECUENCIA.lote_preparado := FALSE;  // el lote ya salió: hay que pulsar "Nuevo lote"
        GVL_SECUENCIA.estado := E_ESTADO.REPOSO;
    END_IF

E_ESTADO.PARO:
    // Amarilla fija. START (con STOP ya suelto) continúa donde iba.
    IF start_pulso AND NOT stop_activo THEN
        GVL_SECUENCIA.estado := estado_previo;
    END_IF

E_ESTADO.FALLA:
    // Roja intermitente. START reconoce la falla si la causa ya desapareció.
    IF start_pulso AND NOT falla_comm AND NOT falla_fc THEN
        GVL_SECUENCIA.falla_codigo := 0;
        GVL_SECUENCIA.estado := E_ESTADO.REPOSO;
    END_IF

E_ESTADO.EMERGENCIA:
    // Roja fija. Hay que desenclavar el pulsador y luego pulsar START.
    // Vuelve a REPOSO (nunca reanuda solo).
    IF NOT emergencia AND start_pulso THEN
        GVL_SECUENCIA.estado := E_ESTADO.REPOSO;
    END_IF

END_CASE

// =====================================================================
// 5. SALIDAS: se calculan a partir del estado (nada queda "pegado")
// =====================================================================
// --- Interfaz con PLC_PRG / Python ---
GVL_SECUENCIA.ventilador := (GVL_SECUENCIA.estado = E_ESTADO.SECADO)
                         OR (GVL_SECUENCIA.estado = E_ESTADO.INVERSION)
                         OR (GVL_SECUENCIA.estado = E_ESTADO.ENFRIAMIENTO);

GVL_SECUENCIA.calentar := (GVL_SECUENCIA.estado = E_ESTADO.SECADO)
                       OR ((GVL_SECUENCIA.estado = E_ESTADO.INVERSION) AND NOT PAUSA_RESIST_EN_INVERSION);

GVL_SECUENCIA.planta_run := GVL_SECUENCIA.ventilador;

// El sentido del aire que recibe Python sale del FINAL DE CARRERA, no del reloj:
// si la compuerta no se mueve, el aire no se invierte.
GVL_SECUENCIA.aire_invertido := compuerta_invertida;

// --- Órdenes de movimiento (la bobina se apaga al llegar al final de carrera) ---
mov_compuerta_inv := (GVL_SECUENCIA.estado = E_ESTADO.INVERSION)
                     AND sentido_pedido AND NOT GVL_SECUENCIA.I_1B2;

mov_compuerta_norm := (((GVL_SECUENCIA.estado = E_ESTADO.INVERSION) AND NOT sentido_pedido)
                       OR (GVL_SECUENCIA.estado = E_ESTADO.CARGA))
                      AND NOT GVL_SECUENCIA.I_1B1;

// La bandeja solo se mueve con el ventilador apagado (CARGA y DESCARGA no lo encienden)
mov_bandeja_adentro := (GVL_SECUENCIA.estado = E_ESTADO.CARGA) AND NOT GVL_SECUENCIA.I_2B2;
mov_bandeja_afuera  := (GVL_SECUENCIA.estado = E_ESTADO.DESCARGA) AND NOT GVL_SECUENCIA.I_2B1;

// --- Bobinas de las válvulas 5/2 biestables (nunca las dos de una válvula a la vez) ---
GVL_SECUENCIA.Q_1S1 := mov_compuerta_inv   AND NOT mov_compuerta_norm;
GVL_SECUENCIA.Q_2S1 := mov_compuerta_norm  AND NOT mov_compuerta_inv;
GVL_SECUENCIA.Q_1S2 := mov_bandeja_adentro AND NOT mov_bandeja_afuera;
GVL_SECUENCIA.Q_2S2 := mov_bandeja_afuera  AND NOT mov_bandeja_adentro;

// --- Ventilador en FluidSIM ---
GVL_SECUENCIA.Q_K1 := GVL_SECUENCIA.ventilador;

// --- Semáforo (IEC 60204-1) ---
tmr_blink(IN := NOT tmr_blink.Q, PT := T#500MS);
IF tmr_blink.Q THEN
    parpadeo := NOT parpadeo;
END_IF

GVL_SECUENCIA.Q_H1_VERDE := (GVL_SECUENCIA.estado = E_ESTADO.CARGA)
                         OR (GVL_SECUENCIA.estado = E_ESTADO.SECADO)
                         OR (GVL_SECUENCIA.estado = E_ESTADO.INVERSION)
                         OR (GVL_SECUENCIA.estado = E_ESTADO.ENFRIAMIENTO)
                         OR (GVL_SECUENCIA.estado = E_ESTADO.DESCARGA)
                         OR ((GVL_SECUENCIA.estado = E_ESTADO.FIN_SECADO) AND parpadeo);

GVL_SECUENCIA.Q_H2_AMARILLA := (GVL_SECUENCIA.estado = E_ESTADO.REPOSO)
                            OR (GVL_SECUENCIA.estado = E_ESTADO.PARO);

GVL_SECUENCIA.Q_H3_ROJA := (GVL_SECUENCIA.estado = E_ESTADO.EMERGENCIA)
                        OR ((GVL_SECUENCIA.estado = E_ESTADO.FALLA) AND parpadeo);

// --- Vigilancia de carrera: se llama en cada ciclo ---
tmr_mov(IN := mov_compuerta_inv OR mov_compuerta_norm OR mov_bandeja_adentro OR mov_bandeja_afuera,
        PT := T_MOV_MAX);

// =====================================================================
// 6. ESCRITURA A FLUIDSIM (OUTPUT0 -> bloque "FluidSIM In")
// =====================================================================
GVL_SECUENCIA.OUTPUT0.0 := GVL_SECUENCIA.Q_1S1;           // 1S1 cilindro 1 sale
GVL_SECUENCIA.OUTPUT0.1 := GVL_SECUENCIA.Q_2S1;           // 2S1 cilindro 1 entra
GVL_SECUENCIA.OUTPUT0.2 := GVL_SECUENCIA.Q_1S2;           // 1S2 cilindro 2 sale
GVL_SECUENCIA.OUTPUT0.3 := GVL_SECUENCIA.Q_2S2;           // 2S2 cilindro 2 entra
GVL_SECUENCIA.OUTPUT0.4 := GVL_SECUENCIA.Q_K1;            // K1 ventilador
GVL_SECUENCIA.OUTPUT0.5 := GVL_SECUENCIA.Q_H1_VERDE;      // lámpara verde
GVL_SECUENCIA.OUTPUT0.6 := GVL_SECUENCIA.Q_H2_AMARILLA;   // lámpara amarilla
GVL_SECUENCIA.OUTPUT0.7 := GVL_SECUENCIA.Q_H3_ROJA;       // lámpara roja

// --- Textos para la HMI ---
CASE GVL_SECUENCIA.estado OF
    E_ESTADO.REPOSO:
        IF GVL_SECUENCIA.lote_preparado THEN
            GVL_SECUENCIA.estado_txt := 'LOTE LISTO - PULSE START';
        ELSE
            GVL_SECUENCIA.estado_txt := 'REPOSO - PULSE NUEVO LOTE';
        END_IF
    E_ESTADO.CARGA:        GVL_SECUENCIA.estado_txt := 'CARGA';
    E_ESTADO.SECADO:       GVL_SECUENCIA.estado_txt := 'SECADO';
    E_ESTADO.INVERSION:    GVL_SECUENCIA.estado_txt := 'INVERSION DEL AIRE';
    E_ESTADO.ENFRIAMIENTO: GVL_SECUENCIA.estado_txt := 'ENFRIAMIENTO';
    E_ESTADO.FIN_SECADO:   GVL_SECUENCIA.estado_txt := 'FIN DE SECADO';
    E_ESTADO.DESCARGA:     GVL_SECUENCIA.estado_txt := 'DESCARGA';
    E_ESTADO.PARO:         GVL_SECUENCIA.estado_txt := 'PARO';
    E_ESTADO.FALLA:        GVL_SECUENCIA.estado_txt := 'FALLA';
    E_ESTADO.EMERGENCIA:   GVL_SECUENCIA.estado_txt := 'EMERGENCIA';
END_CASE

CASE GVL_SECUENCIA.falla_codigo OF
    0: GVL_SECUENCIA.falla_txt := '';
    1: GVL_SECUENCIA.falla_txt := 'Sin comunicacion con Python';
    2: GVL_SECUENCIA.falla_txt := 'Compuerta trabada (1B1/1B2)';
    3: GVL_SECUENCIA.falla_txt := 'Bandeja trabada (2B1/2B2)';
    4: GVL_SECUENCIA.falla_txt := 'Finales de carrera incoherentes';
END_CASE

// =====================================================================
// 7. AUXILIARES DE LA HMI (animaciones y textos que cambian)
// =====================================================================
tmr_anim(IN := NOT tmr_anim.Q, PT := T#100MS);
IF tmr_anim.Q AND GVL_SECUENCIA.Q_K1 THEN
    angulo_vent := angulo_vent + 30.0;
    IF angulo_vent >= 360.0 THEN angulo_vent := angulo_vent - 360.0; END_IF
END_IF
IF GVL_SECUENCIA.I_1B2 THEN angulo_compuerta := 90.0; ELSE angulo_compuerta := 0.0; END_IF
IF GVL_SECUENCIA.I_2B2 THEN x_bandeja := 0; ELSE x_bandeja := -220; END_IF

IF PLC_PRG.comm_error THEN txt_comm := 'ERROR'; ELSE txt_comm := 'OK'; END_IF

IF PLC_PRG.aire_invertido THEN
    txt_aire := 'INVERTIDO (arriba -> abajo)';
ELSE
    txt_aire := 'NORMAL (abajo -> arriba)';
END_IF

IF GVL_SECUENCIA.I_2B2 AND NOT GVL_SECUENCIA.I_2B1 THEN
    txt_bandeja := 'ADENTRO';
ELSIF GVL_SECUENCIA.I_2B1 AND NOT GVL_SECUENCIA.I_2B2 THEN
    txt_bandeja := 'AFUERA';
ELSE
    txt_bandeja := 'EN MOVIMIENTO';
END_IF

IF PLC_PRG.fan_on THEN txt_vent := 'ENCENDIDO'; ELSE txt_vent := 'APAGADO'; END_IF
IF PLC_PRG.heater_on THEN txt_resist := 'ENCENDIDA'; ELSE txt_resist := 'APAGADA'; END_IF

IF GVL_SECUENCIA.estado = E_ESTADO.EMERGENCIA THEN
    txt_alarma := 'PARO DE EMERGENCIA ACTIVO';
ELSIF GVL_SECUENCIA.falla_txt <> '' THEN
    txt_alarma := GVL_SECUENCIA.falla_txt;
ELSE
    txt_alarma := 'Sin alarmas activas';
END_IF
